# Todolist Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The model keeps a list of tasks and the user sees it in a widget over the chat feed.

**Architecture:** The host owns the state. The model calls the tool `todo_write` with the whole list, the tool returns a snapshot in the result, the agent carries the snapshot to the host, the host applies it, sends it to the panel and keeps it in the saved session. The list also goes into the prompt of every turn as the section `Tasks:`.

**Tech Stack:** TypeScript, VS Code extension API, React 18 in the webview, vitest, esbuild.

## Global Constraints

- The texts of the interface are English, the texts that go to the model are English, the documentation of the project is Russian.
- Never use the em dash (U+2014) in the texts of the project.
- The version lives in `package.json`, a new feature raises it by a minor: this plan ends at `0.10.0`.
- The commit message is a Conventional Commit in Russian, the files are staged with explicit paths: `git add -- <path>`, never `git add -A`.
- `npm run typecheck` and `npm test` have to be green before every commit.
- The working tree holds the changes of a parallel session (voice input, composer toolbar): `src/extension.ts`, `src/ui/protocol.ts`, `src/ui/webview/App.tsx`, `src/ui/webview/strings.ts`, `src/ui/webview/styles.css`, `package.json`, `AGENTS.md` and `test/unit/extension-activate.test.ts` are touched by it. Those changes must not land in the commits of this plan. Stage your own hunks only, with the helper:
  `python C:\jtmp\mm-stage.py --repo 'H:\personal_project\minimax-vscode' --file <file> --keep '<a text of your change>' --apply`
  The helper takes the hunks of the file that hold the given text and puts them into the index, the rest of the file stays in the working tree. - `core.autocrlf=true`: the working copy is CRLF, the anchors of a multiline `tools.edit` have to contain `\r\n`. A single line anchor works always.

- The host is the only owner of the state of the session. A tool never changes the state itself, it returns a snapshot.

Create `src/tools/todo.ts`:

```ts
import type {
  TodoItem,
  TodoPriority,
  TodoStatus,
  ToolStateUpdate
} from '../agent/state';
import type { ToolEntry } from './types';

const STATUSES: TodoStatus[] = ['pending', 'in_progress', 'completed'];
const PRIORITIES: TodoPriority[] = ['high', 'medium', 'low'];

// The list is a snapshot: the model sends every task every time and the
// host replaces the old list. A task without a text is dropped, an unknown
// status falls back to pending and an unknown priority to medium.
export function normalizeTodos(input: unknown): TodoItem[] {
  if (!Array.isArray(input)) {
    return [];
  }
  const items: TodoItem[] = [];
  for (const raw of input) {
    if (typeof raw !== 'object' || raw === null) {
      continue;
    }
    const record = raw as Record<string, unknown>;
    const content =
      typeof record['content'] === 'string' ? record['content'].trim() : '';
    if (content.length === 0) {
      continue;
    }
    const status = record['status'] as TodoStatus;
    const priority = record['priority'] as TodoPriority;
    items.push({
      content,
      status: STATUSES.includes(status) ? status : 'pending',
      priority: PRIORITIES.includes(priority) ? priority : 'medium'
    });
  }
  return items;
}

export function createTodoTools(): ToolEntry[] {
  const todoWrite: ToolEntry = {
    name: 'todo_write',
    description:
      'Write the list of tasks of the session. The list is replaced by the new one, so send every task every time. Use it to show the plan and the progress to the user.',
    approval: 'never',
    sideEffect: false,
    inputSchema: {
      type: 'object',
      properties: {
        todos: {
          type: 'array',
          description: 'Every task of the list with its state and its priority.',
          items: {
            type: 'object',
            properties: {
              content: { type: 'string', description: 'What has to be done.' },
              status: {
                type: 'string',
                enum: ['pending', 'in_progress', 'completed'],
                description: 'The state of the task.'
              },
              priority: {
                type: 'string',
                enum: ['high', 'medium', 'low'],
                description: 'How urgent the task is.'
              }
            },
            required: ['content']
          }
        }
      },
      required: ['todos']
    },
    async execute(input, ctx) {
      // The list belongs to the main agent, a sub-agent has its own work.
      if ((ctx.depth ?? 0) > 0) {
        return {
          content:
            'The list of tasks belongs to the main agent. A sub-agent cannot change it.',
          isError: true
        };
      }
      const todos = normalizeTodos(input['todos']);
      const state: ToolStateUpdate = { kind: 'todos', todos };
      const done = todos.filter((item) => item.status === 'completed').length;
      return {
        content:
          'The list of tasks has ' +
          todos.length +
          ' items, ' +
          done +
          ' of them are done.',
        state
      };
    }
  };
  return [todoWrite];
}
```

```ts
// The list of tasks is a section of its own, the model writes it and the

export function renderTasksContext(todos: TodoItem[]): string {
  if (todos.length === 0) {
    return '';
  }
  return [
    'Tasks:',
    ...todos.map(
      (item) => '- [' + item.status + '] ' + item.content + ' (' + item.priority + ')'
    )
  ].join('\n');
}
```

```tsx
import { useState } from 'react';
import type { TodoItem } from '../../../agent/state';
import { IconList } from '../icons';
import { STRINGS } from '../strings';

// The list of tasks over the feed. The model writes the list, the reader
// only looks at it: a click on a task changes nothing.
export function TaskList({ todos }: { todos: TodoItem[] }) {
  const [open, setOpen] = useState(true);
  if (todos.length === 0) {
    return null;
  }
  const done = todos.filter((item) => item.status === 'completed').length;
  return (
    <div className="task-list">
      <button
        className="task-head"
        aria-expanded={open}
        onClick={() => setOpen(!open)}
        title={STRINGS.taskTitle}
      >
        <IconList />
        <span className="task-title">{STRINGS.taskTitle}</span>
        <span className="task-count">
          {done}/{todos.length}
        </span>
      </button>
      {open ? (
        <ul className="task-rows">
          {todos.map((item, index) => (
            <li key={index} className={'task-row task-' + item.status}>
              <span className="task-marker" aria-hidden="true" />
              <span className="task-text">{item.content}</span>
              <span className="task-priority">{item.priority}</span>
            </li>
          ))}
        </ul>
      ) : null}
    </div>
  );
}
```
